Check it yourself

bash docker/quickstart.sh                 # offline: pipeline verify, figures and explorer from results/
python3 scripts/pipeline.py verify        # re-runs the cheap stages and compares them with the shipped files
bash scripts/provision.sh artifacts       # fetches every system's artifact at its pinned commit
python3 explorer/serve.py                 # explorer and labeling app at http://127.0.0.1:8790/
